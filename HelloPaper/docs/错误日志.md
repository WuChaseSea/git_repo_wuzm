```python
with gr.Blocks(
    theme=self._theme,
    css=self._css,
    title=self.app_name,
    analytics_enabled=False,
    # js=self._js,
    head=external_js,
) as demo:
    self.app = demo

    self.ui()

    self.register_events()

    demo.load(None, None, None, js=self._pdf_view_js)

return demo
```
